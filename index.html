-- Record + positional constructor patterns, multi-clause function, record construction
fn addVelocity(Pos, Velocity) -> Pos;

fn addVelocity (Pos px py) (Velocity vx vy) =
    Pos { x = px + vx, y = py + vy };

fn addVelocity pos (Velocity 0 0) =
    pos;

-- if/then/else, comparison, var getter
fn isOrigin(Pos) -> Bool;
fn isOrigin (Pos px py) =
    if px == 0 then True else False;

-- match over a sum type
fn describe(Shape) -> Int;
fn describe shape =
    match shape with
        Circle r -> r,
        Rectangle w h -> w + h;

-- cross-module extern call, record construction as an argument
fn renderOrigin(Renderer) -> Renderer;
fn renderOrigin renderer =
    drawCircle (Pos { x = 0, y = 0 }) renderer;

-- fixed-length list pattern
fn sumFirstTwo(List Int) -> Int;
fn sumFirstTwo pair =
    match pair with
        [a, b] -> a + b,
        _ -> 0;

-- tuples
fn swap((a, b)) -> (b, a);
fn swap (a, b) = (b, a);

-- lambda + closure over an outer var
fn addFive(Int) -> Int;
fn addFive n =
    (.\x -> x + n) 5;

-- generic type instantiation
fn unbox(Box a) -> a;
fn unbox (Box n) = n;

fn boxedArea(Box Shape) -> Int;
fn boxedArea box =
    let shape = unbox box in
    let area =
        match shape with
            Circle r -> r * r,
            Rectangle w h -> w * h
    in
    if area == 0 then 1 else area;

-- record update (Haskell-style: base { field = expr })
fn moveRight(Pos) -> Pos;
fn moveRight pos =
    pos { x = pos.x + 1 };

system MoveSystem #(Pos, Velocity) -> Pos;

system MoveSystem #(pos (Velocity vx vy)) =
    Pos { x = pos.x + vx, y = pos.y + vy };

system RenderSystem #(Pos) -> Pos with res: Renderer;

system RenderSystem #(pos) with renderer =
    pos;
